цільова змінна, а "~ ." - всі інші як предиктори.formula <- as.formula(paste(target_label, "~ ."))# Створення моделі дерева рішень за допомогою rpartrpart_model <- rpart(formula, data = data_train)# Створення моделі дерева рішень за допомогою С50C50_model <- C5.0(formula, data = data_train)# За допомогою моделей прогнозуємо мітку класу в тестовому наборіrpart_predictions <- predict(rpart_model, data_test, type = "class")C50_predictions <- predict(C50_model, data_test)# Оцінюємо точністьrpart_accuracy <- confusionMatrix(data_test[[target_label]], rpart_predictions)C50_accuracy <- confusionMatrix(data_test[[target_label]], C50_predictions)print(rpart_accuracy)print(C50_accuracy)Висновки
