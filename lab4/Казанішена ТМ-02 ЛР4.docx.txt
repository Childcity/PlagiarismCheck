Отже, класифікація за допомогою машин опорних векторів є ефективним методом для розв'язання задач класифікації, особливо коли маємо велику кількість ознак та об'єм даних. При цьому, вибір конкретної бібліотеки залежить від характеристик даних та потреб користувача.Лістинг програмиlibrary(class)
library(caret)

# Завантаження даних
url <- "http://archive.ics.uci.edu/ml/machine-learning-databases/00640/Occupancy_Estimation.csv"
data <- read.csv(url)
# Видаляємо непотрібні колонки
data <- data[, 3:ncol(data)]

data$Room_Occupancy_Count <- as.factor(data$Room_Occupancy_Count)

dependent_var <- "Room_Occupancy_Count"
independent_vars <- setdiff(colnames(data), dependent_var)

set.seed(123)  # Для відтворюваності результатів
train_indices <- createDataPartition(data$Room_Occupancy_Count, p = 0.6, list = FALSE)
train_data <- data[train_indices, ]
test_data <- data[-train_indices, ]


## 3. Провести класифікацію за допомогою машин опорних векторів, використовуючи бібліотеки caret, e1071 та kernlab
## caret
control <- trainControl(method = "cv", number = 10)
metric <- "Accuracy"

fit.svm <- train(Room_Occupancy_Count~., data = train_data, method = "svmLinear", metric = metric, trControl = control)
caret_predict <- predict(fit.svm, train_data)
confusion_matrix <- table(caret_predict, train_data$Room_Occupancy_Count)
print(confusion_matrix)
accuracy1 <- sum(diag(confusion_matrix))/sum(confusion_matrix)
print(paste("accuracy of caret svm =",accuracy1*100,"%"))


############################################################
## e1071
library("e1071")
svm_model <- svm(as.factor(Room_Occupancy_Count) ~ ., data=train_data,
                 kernel="polynomial")
pred = predict(svm_model, test_data)
tab = table(Predicted=pred, Actual = test_data$Room_Occupancy_Count)
print(tab)
accuracy2 <- sum(diag(tab))/sum(tab)
print(paste("accuracy of e1071 svm =",accuracy2*100,"%"))

############################################################
# kernlab

library(kernlab)
classifier <- ksvm(as.factor(Room_Occupancy_Count)~ ., data = train_data, 
                   kernel = "vanilladot")
predictions <- predict(classifier, test_data)
cm <- table(predictions, test_data$Room_Occupancy_Count)
print(cm)
accuracy3 <- sum(diag(cm))/sum(cm)
print(paste("accuracy of ksvm with vanilla kernel =",accuracy3*100,"%"))


classifier <- ksvm(as.factor(Room_Occupancy_Count)~ ., data = train_data, 
                   kernel = "rbfdot")
predictions <- predict(classifier, test_data)
cm <- table(predictions, test_data$Room_Occupancy_Count)
print(cm)
accuracy4 <- sum(diag(cm))/sum(cm)
print(paste("accuracy of ksvm with rbf kernel =",accuracy4*100,"%"))
