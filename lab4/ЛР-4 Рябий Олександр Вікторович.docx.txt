header=TRUE)head(dataset)str(dataset)dataset<-dataset[,c(-1)]dataset<-dataset[,c(-1)]dataset$Room_Occupancy_Count <- as.factor(dataset$Room_Occupancy_Count)library(caret)set.seed(128)dataset.training.indicies <- createDataPartition(dataset$Room_Occupancy_Count, p=0.6, list = FALSE)dataset.training <- dataset[dataset.training.indicies,]dataset.validation <- dataset[-dataset.training.indicies,]table(dataset.training$Room_Occupancy_Count)table(dataset.validation$Room_Occupancy_Count)control <- trainControl(method = "cv", number = 10)metric <- "Accuracy"fit.svm <- train(Room_Occupancy_Count~., data = dataset.training, method = "svmLinear", metric = metric, trControl = control)print(fit.svm)caret_predict <- predict(fit.svm, dataset.validation)confusion_matrix <- table(caret_predict, dataset.validation$Room_Occupancy_Count)print(confusion_matrix)accuracy1 <- sum(diag(confusion_matrix))/sum(confusion_matrix)print(paste("accuracy of caret svm =",accuracy1*100,"%"))library("e1071")svm_model <- svm(Room_Occupancy_Count~., data=dataset.training, sum(diag(tab))/sum(tab)print(paste("accuracy of e1071 svm =",accuracy2*100,"%"))library(kernlab)classifier <- ksvm(Room_Occupancy_Count~., data = dataset.training, kernel = "vanilladot")predictions <- =",accuracy3*100,"%"))classifier <- ksvm(Room_Occupancy_Count ~ ., data = dataset.training, kernel = "rbfdot")predictions <- predict(classifier, dataset.validation)cm <- table(predictions,
