# Класифікація з використанням SVM за допомогою caretcontrol <- trainControl(method = "cv", number = 10)metric <- "Accuracy"fit_svm <- train(V1 ~ ., data = letter_training, method = "svmLinear", metric = metric, trControl = control)caret_predictions <- predict(fit_svm, letter_testing)confusion_matrix_caret <- table(caret_predictions, letter_testing$V1)accuracy_caret <- sum(diag(confusion_matrix_caret)) / sum(confusion_matrix_caret)print(paste("Accuracy of caret SVM:", accuracy_caret * 100, "%"))# Класифікація з використанням SVM за допомогою e1071letter_training$V1 <- as.factor(letter_training$V1)letter_testing$V1 <- as.factor(letter_testing$V1)svm_model <- svm(V1 ~ ., data = letter_training, kernel = "linear")e1071_predictions <- predict(svm_model, letter_testing)confusion_matrix_e1071 <- table(e1071_predictions, letter_testing$V1)accuracy_e1071 <- sum(diag(confusion_matrix_e1071)) / sum(confusion_matrix_e1071)print(paste("Accuracy of e1071 SVM:", accuracy_e1071 * 100, "%"))# Класифікація з використанням SVM за допомогою kernlab (vanilla kernel)classifier_vanilla <- ksvm(V1 ~ ., data = letter_training, kernel = "vanilladot")predictions_vanilla <- predict(classifier_vanilla, letter_testing)confusion_matrix_vanilla <- table(predictions_vanilla, letter_testing$V1)accuracy_vanilla <- sum(diag(confusion_matrix_vanilla)) / sum(confusion_matrix_vanilla)print(paste("Accuracy of ksvm with vanilla kernel:", accuracy_vanilla * 100, "%"))# Класифікація з використанням SVM за допомогою kernlab (rbf kernel)classifier_rbf <- ksvm(V1 ~ ., data = letter_training, kernel = "rbfdot")predictions_rbf <- predict(classifier_rbf, letter_testing)confusion_matrix_rbf <- table(predictions_rbf, letter_testing$V1)accuracy_rbf <- sum(diag(confusion_matrix_rbf)) / sum(confusion_matrix_rbf)print(paste("Accuracy of ksvm with rbf kernel:", accuracy_rbf * 100, "%"))5.
