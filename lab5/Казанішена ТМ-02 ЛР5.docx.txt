Висновки: Метою було побудова моделі для класифікації даних про рак молочної залози на основі вхідних ознак з метою передбачення доброякісних (B) або злоякісних (M) пухлин.Отримані результати показують, що модель логістичної регресії має високу точність в класифікації раку молочної залози. Навчальна точність та тестова точність підтверджують, що модель здатна ефективно передбачати класи на нових даних. ROC-крива та показник AUC підтверджують, що модель має хорошу роздільну здатність між класами.Загалом, логістична регресія є потужним інструментом для класифікації даних про рак молочної залози на основі вхідних ознак. Результати роботи можуть бути використані для допомоги в ранньому виявленні та діагностиці раку молочної залози з високою точністю.Повний лістинг програмиlibrary(tidyverse)library(caret)library(pROC)# 1. Імпортувати даніurl <- "http://archive.ics.uci.edu/ml/machine-learning-databases/breast-cancer-wisconsin/wdbc.data"breast_cancer <- read.csv(url, header = FALSE)breast_cancer$V2 <- ifelse(breast_cancer$V2 == "M", 1, 0)# 2. Розбити набір даних на навчальний та тестовийset.seed(42)train_idx <- sample(nrow(breast_cancer), 0.7 * nrow(breast_cancer))train_data <- breast_cancer[train_idx, ]test_data <- breast_cancer[-train_idx, ]# 3. Провести класифікацію за допомогою логістичної регресіїmodel_glm <- glm(V2 ~ ., data = train_data, family = binomial)head(predict(model_glm))head(predict(model_glm, type = "link"))head(predict(model_glm, type = "response"))# 5. Оцінити та порівняти точністьtrain_predictions <- predict(model_glm, newdata = train_data, type = "response")train_class_predictions <- ifelse(train_predictions > 0.5, 1, 0)train_accuracy <- mean(train_class_predictions == train_data$V2)test_predictions <- predict(model_glm, newdata = test_data, type = "response")test_class_predictions <- ifelse(test_predictions > 0.5, 1, 0)test_accuracy <- mean(test_class_predictions == test_data$V2)# 6. Побудувати ROC-криву та обчислити показник AUCroc_obj <- roc(test_data$V2, test_predictions)auc <- auc(roc_obj)plot(roc_obj, main = "ROC Curve", col = "blue")cat("Навчальна точність:", train_accuracy, "\n")cat("Тестова точність:", test_accuracy, "\n")cat("AUC:", auc, "\n")
